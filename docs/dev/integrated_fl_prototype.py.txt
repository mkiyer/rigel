"""PROTOTYPE: the INTEGRATED gDNA fl estimator -- regions and boundaries in ONE solve.

THE DESIGN (owner direction, 2026-08-31): capture is a SPECTRUM. The tool never knows whether capture is
on, or how strong. So there is no regime switch anywhere:

  stage 1  rho_off + contained weights + contained contrast        (what ships today, unchanged)
  stage 2  per-boundary composition, with the REGIONS calibrating the BOUNDARIES:
             probes bind nucleic acid indiscriminately, so at one boundary gDNA and nascent are
             enriched at the SAME rate and the enrichment CANCELS from the composition:
                 R_b  = (rho_r,adj / rho_off) * E_r[L-1] / E_g[L-1]     (RNA:gDNA odds at boundary b)
                 a_b  = 1 / (1 + R_b)
             where rho_r,adj is the ADJACENT non-exon region's own RNA excess density,
             measured as max(0, n_i - rho_off*E_g,i) / E_r,i  -- the same one-sided residual again.
           crossing weights a2, a3 = count-weighted a_b over each boundary class; crossing contrast.
  stage 3  blend by REALIZED gDNA MASS:  g = (m_C*g_C + m_B*g_B)/(m_C+m_B),
             m_C = a0*n0 + a1*n1,  m_B = a2*n2 + a3*n3.
           The masses ARE the capture knob: capture off -> m_B is negligible and this IS the shipped
           estimator; capture strong -> m_C empties and the boundaries take over. Nothing switches.

Scored against the origin-split oracle beside the SHIPPED estimator, across every substrate.
"""

CROSS = (POOL_DNA_INTRON_EXON, POOL_DNA_INTERGENIC_EXON)


def mean_of(p):
    s = p.sum()
    return float((p * np.arange(p.size)).sum() / s) if s > 0 else float("nan")


def tv(a, b):
    if a.sum() <= 0 or b.sum() <= 0:
        return float("nan")
    return 0.5 * float(np.abs(a / a.sum() - b / b.sum()).sum())


def strand_sum(a):
    a = np.asarray(a, dtype=np.float64)
    return a.sum(axis=1) if a.ndim > 1 else a


def detilted_pool(payload, opp, pool):
    total = np.asarray(opp.total, dtype=np.float64)
    A = np.asarray(opp.pools[pool], dtype=np.float64)
    prob = np.zeros_like(A)
    np.divide(A, total, out=prob, where=total > 0.0)
    return _normalized(detilt_pool(np.asarray(payload.pool_lengths, dtype=np.float64)[pool], prob))


def isotonic_increasing(y, w):
    """Weighted PAVA. Bins vote with their weights, so empty bins carry no influence and the fit
    extends FLAT where there is no data -- a monotone curve cannot explode past its last evidence."""
    y = np.asarray(y, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    vals, wts, runs = [], [], []
    for yi, wi in zip(y, np.maximum(w, 0.0)):
        vals.append(yi if wi > 0 else (vals[-1] if vals else 0.0))
        wts.append(wi)
        runs.append(1)
        while len(vals) > 1 and vals[-2] > vals[-1]:
            tot = wts[-1] + wts[-2]
            vals[-2] = (vals[-1] * wts[-1] + vals[-2] * wts[-2]) / tot if tot > 0 else vals[-2]
            wts[-2] = tot
            runs[-2] += runs[-1]
            vals.pop(); wts.pop(); runs.pop()
    out = np.empty(y.size, dtype=np.float64)
    at = 0
    for v, r in zip(vals, runs):
        out[at:at + r] = v
        at += r
    return np.maximum(out, 0.0)


def contrast(f_a, f_b, a_a, a_b):
    g = ((1.0 - a_b) * f_a - (1.0 - a_a) * f_b) / (a_a - a_b)
    return _normalized(np.clip(g, 0.0, None))


def integrated(payload, index, opp, region_lengths, region_types, ref_region_offsets,
               ref_boundary_offsets, rna_pmf, iters=2):
    """Returns (pmf | None, diagnostics dict)."""
    size = int(payload.max_length)
    obs_pools = np.asarray(payload.pool_lengths, dtype=np.float64)
    ty = np.asarray(region_types).ravel()
    ell = np.asarray(region_lengths, dtype=np.float64).ravel()
    cnt = strand_sum(payload.region_contained_count)
    n = min(ty.size, ell.size, cnt.size)
    ty, ell, cnt = ty[:n], ell[:n], cnt[:n]
    is_ig = (ty == int(RegionType.INTERGENIC)) & (ell > 0)
    is_in = (ty == int(RegionType.INTRON)) & (ell > 0)

    # ── stage 1: the shipped machinery, reused verbatim
    ship_counts, diag = _deconvolved_gdna_counts(payload, opp, region_lengths, region_types)
    pooled_pmf = _normalized(obs_pools[list(_GDNA_CONTAINED_POOLS)].sum(axis=0))
    E_g_regions = contained_opportunity(pooled_pmf, ell)
    fit = one_sided_rate(cnt[is_ig | is_in], E_g_regions[is_ig | is_in])
    if not fit.informative:
        return None, {"declined": "no gDNA density"}
    rho_off = fit.rate
    a_cont, n_cont = {}, {}
    for lab, m in (("ig", is_ig), ("in", is_in)):
        n_p = float(cnt[m].sum())
        a_cont[lab] = min(rho_off * float(E_g_regions[m].sum()) / max(n_p, 1e-9), 1.0)
        n_cont[lab] = n_p
    g_C = _normalized(ship_counts) if ship_counts is not None else pooled_pmf
    m_C = a_cont["ig"] * n_cont["ig"] + a_cont["in"] * n_cont["in"]

    # ── stage 2: per-boundary composition (regions calibrate boundaries; enrichment cancels)
    r_pmf = np.asarray(rna_pmf, dtype=np.float64)
    mu_r = mean_of(r_pmf)
    # per-region RNA excess density, in the RNA pmf's own opportunity units
    E_r_regions = contained_opportunity(r_pmf[: size + 1], ell)
    excess = np.clip(cnt - rho_off * E_g_regions, 0.0, None)
    rho_r = np.zeros_like(excess)
    np.divide(excess, E_r_regions, out=rho_r, where=E_r_regions > 0.0)

    bnd_cnt = strand_sum(payload.boundary_unspliced_count)
    roff = np.asarray(ref_region_offsets, dtype=np.int64)
    boff = np.asarray(ref_boundary_offsets, dtype=np.int64)

    g_B, m_B, a_cross, cls_counts = None, 0.0, {}, {}
    mu_g = mean_of(g_C)  # the fixed point's seed; refined below
    L_axis = np.arange(g_C.size, dtype=np.float64)
    E1_phi = float((g_C * np.clip(L_axis - 1.0, 0.0, None)).sum())
    E2_phi = float((g_C * L_axis * np.clip(L_axis - 1.0, 0.0, None)).sum()) / 2.0
    exon_eps = {}
    for _ in range(iters):
        num = {2: 0.0, 3: 0.0}
        den = {2: 0.0, 3: 0.0}
        exon_eps = {}
        for r in range(roff.size - 1):
            r_lo, r_hi = int(roff[r]), int(roff[r + 1])
            b_lo, b_hi = int(boff[r]), int(boff[r + 1])
            k = r_hi - r_lo
            if k < 2 or b_hi - b_lo != k - 1:
                continue
            for j in range(k - 1):
                left, right = r_lo + j, r_lo + j + 1
                tl, tr = int(ty[left]), int(ty[right])
                exon = int(RegionType.EXON)
                if tl == exon and tr != exon:
                    adj, other = right, tr
                elif tr == exon and tl != exon:
                    adj, other = left, tl
                else:
                    continue
                cls = 2 if other == int(RegionType.INTRON) else 3
                exon_region = left if tl == exon else right
                nb = float(bnd_cnt[b_lo + j])
                if nb <= 0.0:
                    continue
                R_b = (rho_r[adj] / max(rho_off, 1e-30)) * max(mu_r - 1.0, 1e-9) / max(
                    mu_g - 1.0, 1e-9
                )
                a_b = 1.0 / (1.0 + R_b)
                num[cls] += a_b * nb
                den[cls] += nb
                # the boundary's enrichment RATIO over the off-target rate: 1 = uniform. SIGNED and
                # unclipped here -- the clip lives at the exon mean, so noise cancels instead of
                # accumulating one-sidedly.
                eps_b = (a_b * nb) / max(rho_off * max(mu_g - 1.0, 1e-9), 1e-30)
                exon_eps.setdefault(exon_region, []).append(eps_b)
        if den[2] <= 0.0 or den[3] <= 0.0:
            break
        a2, a3 = num[2] / den[2], num[3] / den[3]
        n2, n3 = float(obs_pools[CROSS[0]].sum()), float(obs_pools[CROSS[1]].sum())
        a_cross, cls_counts = {"a2": a2, "a3": a3}, {"n2": n2, "n3": n3}
        f2, f3 = detilted_pool(payload, opp, CROSS[0]), detilted_pool(payload, opp, CROSS[1])
        sep = a2 - a3
        # ⭐ THE COMPLEMENT-FIRST SOLVE — the same 2x2 system, in the order that conditions it.
        # Solve for the CONTAMINANT r first, clip it to the cone (a density cannot be negative — the
        # physical constraint is the regularizer), and subtract it from the count-weighted mixture:
        #
        #     r_hat = clip( [a3*f2 - a2*f3] / (a3 - a2) )
        #     g_B   = clip( f_mix - (1 - a_mix)*r_hat ) / a_mix
        #
        # In exact arithmetic this IS the pair inversion; in noise it is better conditioned exactly
        # where the inversion is worst: r_hat's 1/sep amplification enters g_B damped by (1 - a_mix),
        # which is small precisely when the two pools are both nearly pure and sep is small. No
        # switch, no threshold — the damping is the algebra's own.
        if abs(sep) > np.sqrt(a2 * a2 / max(n2, 1.0) + a3 * a3 / max(n3, 1.0)):
            a_mix = (a2 * n2 + a3 * n3) / max(n2 + n3, 1.0)
            f_mix = _normalized(n2 * f2 + n3 * f3)
            r_hat = _normalized(np.clip((a3 * f2 - a2 * f3) / (a3 - a2), 0.0, None))
            g_B = _normalized(np.clip(f_mix - (1.0 - a_mix) * r_hat, 0.0, None))
        else:
            g_B = _normalized(a2 * n2 * f2 + a3 * n3 * f3)
        m_B = a2 * n2 + a3 * n3
        mu_g_next = mean_of(g_B)  # the boundary formula wants the BOUNDARY stratum's gDNA mean
        if not np.isfinite(mu_g_next) or abs(mu_g_next - mu_g) < 0.25:
            mu_g = mu_g_next if np.isfinite(mu_g_next) else mu_g
            break
        mu_g = mu_g_next

    if g_B is None or m_B <= 0.0:
        out = ship_counts if ship_counts is not None else None
        return out, {"declined": "no boundary estimate", "m_C": m_C, **a_cont}

    # ── stage 3: THE EXCESS-ENRICHMENT CORRECTION, and the principle that makes it close.
    # The sampled union (contained pair + crossing pair, each deconvolved and mass-weighted) is an
    # unbiased estimate of the library gDNA pmf UNDER UNIFORM PLACEMENT -- the uniform part of every
    # unsampled class is already represented in it. What uniform sampling cannot see is capture's
    # DEVIATION from uniformity, and that deviation is measured: each exon's boundaries report their
    # enrichment ratio eps over the off-target rate. So the unsampled on-target classes (contained-in-
    # exon, spanning-the-exon) enter weighted by the EXCESS only, (eps_e - 1)+:
    #
    #     g  ∝  m_C·g_C  +  m_B·g_B  +  rho_off · Σ_e (eps_e - 1)+ · phi(L)·[ (ell_e - L + 1)+ + (L - ell_e - 1)+ ]
    #
    # At eps = 1 (no capture) the correction vanishes IDENTICALLY and this is the shipped estimator.
    # Under capture eps is measured in the hundreds and the correction carries the on-target mass. No
    # switch, no constant; the exon-mean of SIGNED per-boundary ratios cancels noise, and the clip
    # sits at the physically one-sided quantity (enrichment below baseline is depletion the off-target
    # terms already carry).
    is_ex = (ty == int(RegionType.EXON)) & (ell > 0)
    mean_eps = (float(np.mean([np.mean(v) for v in exon_eps.values()])) if exon_eps else 1.0)
    h_E = np.zeros_like(g_C)
    m_E = 0.0
    # ⭐ EACH CLASS GETS THE LOCAL LAW ITS OWN OVERLAP GEOMETRY DICTATES. Hybridization scales with
    # the bases available to the probe, so the capture length-tilt a class feels is set by how its
    # probe overlap grows with fragment length:
    #   SPANNING the exon: overlap = ell, CONSTANT in L  -> no length tilt -> local law phi (g_C).
    #   CONTAINED in the exon: overlap = L, grows with L -> the measured tilt applies -> local law
    #     g_B, the deconvolved boundary pmf, which carries exactly that hybridization tilt (its
    #     crossing geometry was divided out; the capture tilt was not).
    # Off capture the correction vanishes with (eps-1)+ and the distinction dissolves.
    # ⚠ THE ONE KNOWN RESIDUAL, measured both ways: the contained class's law should carry the
    # within-exon hybridization tilt w(L). Using g_B as its proxy fixes big-exon substrates (test
    # chromosome ON: -8 -> +1) and breaks small-exon ones (ladder ON: +7 -> +20..+45), because a
    # small exon's crossers overlap weakly and g_B then measures a different part of the binding
    # curve. phi is the conservative choice that meets the match-or-improve bar EVERYWHERE; the
    # binding curve itself is the principled fix and needs more statistics than one boundary pool.
    law_contained = g_C
    if exon_eps and law_contained is not None:
        mlaw = min(law_contained.size, g_C.size, L_axis.size)
        for e_idx in np.flatnonzero(is_ex):
            excess_e = max(float(np.mean(exon_eps.get(int(e_idx), [mean_eps]))) - 1.0, 0.0)
            if excess_e <= 0.0:
                continue
            le = ell[e_idx]
            shape = (law_contained[:mlaw] * np.clip(le - L_axis[:mlaw] + 1.0, 0.0, None)
                     + g_C[:mlaw] * np.clip(L_axis[:mlaw] - le - 1.0, 0.0, None))
            h_E[:mlaw] += rho_off * excess_e * shape
            m_E += rho_off * excess_e * float(shape.sum())
    total_mass = m_C + m_B + m_E
    if total_mass <= 0.0:
        return (ship_counts if ship_counts is not None else None,
                {"declined": "no gDNA mass anywhere"})
    m0 = min(g_C.size, (g_B.size if g_B is not None else g_C.size), h_E.size)
    g = m_C * g_C[:m0] + (m_B * g_B[:m0] if g_B is not None else 0.0) + h_E[:m0]
    g = _normalized(g) * float(obs_pools[list(_GDNA_POOLS)].sum())
    return g, {"m_C": m_C, "m_B": m_B, "m_E": m_E, "share_B": m_B / total_mass,
               "share_E": m_E / total_mass, "mean_eps": mean_eps, **a_cont, **a_cross,
               "rho_off": rho_off, "shipped_declined": ship_counts is None}

def main() -> int:
    panel, idx = Path(sys.argv[1]), sys.argv[2]
    conds = sys.argv[3:]
    index = TranscriptIndex.load(idx)
    b, off, rt = build_region_partition_arrays(index)
    rl = region_lengths_from_partition(b, off, len(rt))
    print(f"{'condition':36s} {'TRUE':>7s} {'SHIP':>7s} {'INTEG':>7s} | {'ship err':>8s} {'integ err':>9s} "
          f"| {'shipTV':>7s} {'intgTV':>7s} | {'B-share':>7s} {'a2':>5s} {'a3':>5s}")
    for cond in conds:
        pl = read_scan_cache(panel / "scan_cache" / cond, index).payload
        size = int(pl.max_length)
        gdir = panel / "oracle_cache" / cond / "gdna"
        if not gdir.exists():
            continue
        gt_counts = np.asarray(read_scan_cache(gdir, index).payload.deposited_lengths, np.float64)
        if gt_counts.sum() == 0:
            print(f"{cond.replace('_nrna_file','').replace('_nrna_mid',''):36s}  zero gDNA")
            continue
        gt = gt_counts / gt_counts.sum()
        opp = gdna_opportunity_from_index(index, size)
        sj = crossing_probability_from_index(index, size)
        models = build_fl_models(pl, sj_opportunity=sj, gdna_opportunity=opp,
                                 region_lengths=rl, region_types=rt)
        ship = _normalized(models.gdna_counts)
        pp = read_scan_cache(panel / "scan_cache" / cond, index).payload
        g_int, d = integrated(pp, index, opp, rl, rt, pp.ref_region_offsets,
                              pp.ref_boundary_offsets, models.rna_pmf)
        name = cond.replace("_nrna_file", "").replace("_nrna_mid", "")
        if g_int is None:
            print(f"{name:36s}  DECLINED: {d.get('declined')}")
            continue
        gi = _normalized(g_int)
        m = min(gt.size, ship.size, gi.size)
        gt_, ship_, gi_ = gt[:m], ship[:m], gi[:m]
        print(f"{name:36s} {mean_of(gt_):7.2f} {mean_of(ship_):7.2f} {mean_of(gi_):7.2f} | "
              f"{mean_of(ship_) - mean_of(gt_):+8.2f} {mean_of(gi_) - mean_of(gt_):+9.2f} | "
              f"{tv(ship_, gt_):7.4f} {tv(gi_, gt_):7.4f} | {d.get('share_B', 0):7.3f} E={d.get('share_E', 0):5.3f} "
              f"{d.get('a2', float('nan')):5.2f} {d.get('a3', float('nan')):5.2f}")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
